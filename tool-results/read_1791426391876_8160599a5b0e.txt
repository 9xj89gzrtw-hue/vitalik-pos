    73→function persistJson(key: string, value: unknown) {
    74→  try {
    75→    localStorage.setItem(key, JSON.stringify(value))
    76→  } catch {
    77→    /* no-op */
    78→  }
    79→}
    80→function readJson<T>(key: string): T | undefined {
    81→  try {
    82→    const raw = localStorage.getItem(key)
    83→    if (raw) return JSON.parse(raw) as T
    84→    const legacy = LEGACY_KEYS[key]
    85→    if (legacy) {
    86→      const old = localStorage.getItem(legacy)
    87→      if (old) return JSON.parse(old) as T
    88→    }
    89→    return undefined
    90→  } catch {
    91→    return undefined
    92→  }
    93→}

    96→function sanitizeStopList(raw: unknown): StopList {
    97→  const out: StopList = {}
    98→  if (!raw || typeof raw !== 'object' || Array.isArray(raw)) return out
    99→  for (const [id, value] of Object.entries(raw as Record<string, unknown>)) {
   100→    if (!findMenuItem(id) || !value || typeof value !== 'object') continue
   101→    const v = value as { stopped?: unknown; remaining?: unknown }
   102→    const stopped = v.stopped === true
   103→    const remNum = Number(v.remaining)
   104→    const remaining =
   105→      v.remaining === null || v.remaining === undefined
   106→        ? null
   107→        : Number.isFinite(remNum)
   108→          ? Math.max(0, Math.min(999, Math.floor(remNum)))
   109→          : null
   110→    if (!stopped && remaining === null) continue
   111→    out[id] = { stopped, remaining }
   112→  }
   113→  return out
   114→}

   191→function draftKeyOf(item: { menuItemId: string; garnishId?: string; standalone?: boolean }): string {
   192→  return `${item.menuItemId}::${item.garnishId ?? ''}::${item.standalone ? 1 : 0}`

   195→function sanitizeDrafts(raw: unknown): Record<string, TableDraft> {
   196→  const out: Record<string, TableDraft> = {}
   197→  if (!raw || typeof raw !== 'object') return out
   198→  for (const [tableId, value] of Object.entries(raw as Record<string, TableDraft>)) {
   199→    if (!VALID_TABLE_IDS.has(tableId) || !value || !Array.isArray(value.items)) continue
   200→    const items = value.items
   201→      .filter((c) => c && typeof c.menuItemId === 'string' && findMenuItem(c.menuItemId))
   202→      .map((c) => {
   203→        const menuItem = findMenuItem(c.menuItemId)!
   204→        const garnishId =
   205→          typeof c.garnishId === 'string' &&
   206→          GARNISH_IDS.has(c.garnishId) &&
   207→          menuItem.category !== 'ГАРНИРЫ'
   208→            ? c.garnishId
   209→            : undefined
   210→        return {
   211→          key: draftKeyOf({ menuItemId: c.menuItemId, garnishId, standalone: c.standalone }),
   212→          menuItemId: c.menuItemId,
   213→          garnishId,
   214→          name: typeof c.name === 'string' ? c.name : (menuItem?.name ?? c.menuItemId),
   215→          qty: Math.max(1, Math.min(50, Math.floor(Number(c.qty) || 1))),
   216→          comment: typeof c.comment === 'string' ? c.comment.slice(0, 80) : undefined,
   217→          standalone: c.standalone === true,
   218→        }
   219→      })
   220→    out[tableId] = {
   221→      items,
   222→      vip: value.vip === true,
   223→      tableNote: typeof value.tableNote === 'string' ? value.tableNote.slice(0, 140) : '',
   224→    }
   225→  }
   226→  return out
   227→}

   229→function sanitizeOutbox(raw: unknown): OutboxEntry[] {
   230→  if (!Array.isArray(raw)) return []
   231→  return raw.filter(
   232→    (e): e is OutboxEntry =>
   233→      e &&
   234→      typeof e === 'object' &&
   235→      typeof e.clientOrderId === 'string' &&
   236→      VALID_TABLE_IDS.has(String(e.tableId)) &&
   237→      Array.isArray(e.items) &&
   238→      e.items.length > 0,
   239→  )
   240→}

   242→function persistOutbox(outbox: OutboxEntry[]) {
   243→  persistJson(LS.outbox, outbox)

   516→      items: draft.items.map((c) => (c.key === key ? { ...c, comment: trimmed || undefined } : c)),

   521→
   522→  toggleDraftQuickNote: (tableId, key, note) => {
   523→    const drafts = { ...get().drafts }
   524→    const draft = drafts[tableId]
   525→    if (!draft) return
   526→    drafts[tableId] = {
   527→      ...draft,
   528→      items: draft.items.map((c) => {
   529→        if (c.key !== key) return c
   530→        const parts = (c.comment ?? '')
   531→          .split(', ')
   532→          .map((s) => s.trim())
   533→          .filter(Boolean)
   534→        const idx = parts.indexOf(note)
   535→        if (idx >= 0) parts.splice(idx, 1)
   536→        else parts.push(note)
   537→        const merged = parts.join(', ')
   538→        return { ...c, comment: merged || undefined }
   539→      }),
   540→    }
   541→    persistJson(LS.drafts, drafts)

   570→  sending: false,

   572→  submitDraft: async (tableId) => {
   573→    const state = get()
   574→    const draft = state.drafts[tableId]
   575→    if (!draft || draft.items.length === 0 || state.sending) return false
   576→    if (!state.waiterName.trim()) {
   577→      toast.error('Выберите имя официанта', { description: 'Кухня должна знать, чей заказ' })
   578→      return false
   579→    }
   580→
   581→    // стоп-лист и остатки: сверка ДО отправки (суммарно по блюду + привязанным гарнирам)
   582→    const needed = new Map<string, number>()
   583→    for (const c of draft.items) {
   584→      needed.set(c.menuItemId, (needed.get(c.menuItemId) ?? 0) + c.qty)
   585→      if (c.garnishId) needed.set(c.garnishId, (needed.get(c.garnishId) ?? 0) + c.qty)
   586→    }
   587→    const violations: string[] = []
   588→    for (const [id, qty] of needed) {
   589→      const control = state.stopList[id]
   590→      const name = findMenuItem(id)?.name ?? id
   591→      if (control?.stopped) {
   592→        violations.push(`«${name}» — в стоп-листе`)
   593→      } else if (control?.remaining != null && qty > control.remaining) {
   594→        violations.push(`«${name}» — осталось ${control.remaining}, в чеке ${qty}`)
   595→      }
   596→    }
   597→    if (violations.length) {
   598→      haptic([25, 60, 25])
   599→      toast.error('Заказ не отправлен — стоп-лист', {
   600→        description: violations.slice(0, 3).join(' · '),
   601→      })
   602→      return false
   603→    }
   604→
   605→    const pieces = draft.items.reduce((acc, c) => acc + c.qty, 0)
   606→    const isAddendum = state.orders.some((o) => o.tableId === tableId && o.status !== 'served')
   607→    const tableLabel =
   608→      state.orders.find((o) => o.tableId === tableId)?.tableLabel ?? tableLabelOfSafe(tableId)
   609→
   610→    const payload: SubmitPayload = {
   611→      clientOrderId: crypto.randomUUID(),
   612→      tableId,
   613→      waiterName: state.waiterName.trim(),
   614→      isVIP: draft.vip,
   615→      tableNote: draft.tableNote || undefined,
   616→      period: state.period,
   617→      items: draft.items.map(({ menuItemId, garnishId, qty, comment, standalone }) => ({
   618→        menuItemId,
   619→        garnishId,
   620→        qty,
   621→        comment,
   622→        standalone,
   623→      })),
   624→    }
   625→
   626→    const queueOffline = () => {
   627→      const outbox = [...get().outbox, { ...payload, queuedAt: Date.now(), tableLabel, pieces }]
   628→      persistOutbox(outbox)
   629→      set({ outbox, waiterTab: 'orders' })
   630→      get().clearDraft(tableId)
   631→      toast.warning('Нет связи с кухней', {
   632→        description: 'Заказ сохранён в телефоне и уйдёт автоматически при подключении',
   633→      })
   634→    }
   635→
   636→    set({ sending: true })
   637→    try {
   638→      if (state.connection !== 'online') {
   639→        queueOffline()
   640→        return true
   641→      }
   642→      const res = await emitAck<SubmitAck>('vk:order:submit', payload)
   643→      if (res?.ok) {
   644→        get().clearDraft(tableId)
   645→        playSendConfirm()
   646→        haptic([14, 50, 24])
   647→        toast.success(
   648→          isAddendum ? 'Дозаказ отправлен на кухню' : 'Заказ отправлен! 🟡 Ждёт подтверждения шефа',
   649→          { description: `${tableLabel} · ${pluralDishes(pieces)}${draft.vip ? ' · ⚡ ВИП' : ''}` },
   650→        )
   651→        set({ waiterTab: 'orders' })
   652→        return true
   653→      }
   654→      toast.error('Кухня не приняла заказ', { description: res?.error ?? 'Попробуйте ещё раз' })
   655→      return false
   656→    } catch {
   657→      queueOffline()
   658→      return true
   659→    } finally {
   660→      set({ sending: false })
   661→    }
   662→  },
   663→
   664→  outbox: [],
   665→
   666→  cancelOutboxEntry: (clientOrderId) => {
   667→    const outbox = get().outbox.filter((e) => e.clientOrderId !== clientOrderId)
   668→    persistOutbox(outbox)
   669→    set({ outbox })

   672→  flushOutbox: async () => {
   673→    const state = get()
   674→    if (!state.outbox.length) return
   675→    const remaining = [...state.outbox]
   676→    for (const entry of remaining) {
   677→      try {
   678→        const res = await emitAck<SubmitAck>('vk:order:submit', entry)
   679→        if (res?.ok) {
   680→          const outbox = get().outbox.filter((e) => e.clientOrderId !== entry.clientOrderId)
   681→          persistOutbox(outbox)
   682→          set({ outbox })
   683→          toast.success(`Заказ доставлен на кухню · ${entry.tableLabel}`, {
   684→            description: 'Связь восстановлена — заказ ушёл из офлайн-очереди',
   685→          })
   686→        } else {
   687→          // сервер отказал (некорректные данные) — убираем, чтобы не крутился вечно
   688→          const outbox = get().outbox.filter((e) => e.clientOrderId !== entry.clientOrderId)
   689→          persistOutbox(outbox)
   690→          set({ outbox })
   691→          toast.error(`Заказ отклонён · ${entry.tableLabel}`, { description: res?.error })
   692→        }
   693→      } catch {
   694→        break // связь снова пропала — попробуем при следующем подключении
   695→      }
   696→    }
   697→  },

   791→/* локальный безопасный лейбл стола (без циклического импорта меню в рантайме) */
   792→function tableLabelOfSafe(tableId: string): string {
   793→  const m = /^t(\d+)$/.exec(tableId)
   794→  if (m) return `Стол ${m[1]}`
   795→  if (tableId === 'banquet1') return 'Банкет 1'
   796→  if (tableId === 'banquet2') return 'Банкет 2'
   797→  return tableId
   798→}
