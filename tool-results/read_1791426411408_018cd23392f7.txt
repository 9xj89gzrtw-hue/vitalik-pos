    73→    73→function persistJson(key: string, value: unknown) {
    74→    74→  try {
    75→    75→    localStorage.setItem(key, JSON.stringify(value))
    76→    76→  } catch {
    77→    77→    /* no-op */
    78→    78→  }

    80→    80→function readJson<T>(key: string): T | undefined {
    81→    81→  try {
    82→    82→    const raw = localStorage.getItem(key)
    83→    83→    if (raw) return JSON.parse(raw) as T
    84→    84→    const legacy = LEGACY_KEYS[key]
    85→    85→    if (legacy) {
    86→    86→      const old = localStorage.getItem(legacy)
    87→    87→      if (old) return JSON.parse(old) as T
    88→    88→    }
    89→    89→    return undefined
    90→    90→  } catch {
    91→    91→    return undefined
    92→    92→  }
    93→    93→}

    96→    96→function sanitizeStopList(raw: unknown): StopList {
    97→    97→  const out: StopList = {}
    98→    98→  if (!raw || typeof raw !== 'object' || Array.isArray(raw)) return out
    99→    99→  for (const [id, value] of Object.entries(raw as Record<string, unknown>)) {
   100→   100→    if (!findMenuItem(id) || !value || typeof value !== 'object') continue
   101→   101→    const v = value as { stopped?: unknown; remaining?: unknown }
   102→   102→    const stopped = v.stopped === true
   103→   103→    const remNum = Number(v.remaining)
   104→   104→    const remaining =
   105→   105→      v.remaining === null || v.remaining === undefined
   106→   106→        ? null
   107→   107→        : Number.isFinite(remNum)
   108→   108→          ? Math.max(0, Math.min(999, Math.floor(remNum)))
   109→   109→          : null
   110→   110→    if (!stopped && remaining === null) continue
   111→   111→    out[id] = { stopped, remaining }
   112→   112→  }
   113→   113→  return out
   114→   114→}

   195→   195→function sanitizeDrafts(raw: unknown): Record<string, TableDraft> {
   196→   196→  const out: Record<string, TableDraft> = {}
   197→   197→  if (!raw || typeof raw !== 'object') return out
   198→   198→  for (const [tableId, value] of Object.entries(raw as Record<string, TableDraft>)) {
   199→   199→    if (!VALID_TABLE_IDS.has(tableId) || !value || !Array.isArray(value.items)) continue
   200→   200→    const items = value.items
   201→   201→      .filter((c) => c && typeof c.menuItemId === 'string' && findMenuItem(c.menuItemId))
   202→   202→      .map((c) => {
   203→   203→        const menuItem = findMenuItem(c.menuItemId)!
   204→   204→        const garnishId =
   205→   205→          typeof c.garnishId === 'string' &&
   206→   206→          GARNISH_IDS.has(c.garnishId) &&
   207→   207→          menuItem.category !== 'ГАРНИРЫ'
   208→   208→            ? c.garnishId
   209→   209→            : undefined
   210→   210→        return {
   211→   211→          key: draftKeyOf({ menuItemId: c.menuItemId, garnishId, standalone: c.standalone }),
   212→   212→          menuItemId: c.menuItemId,
   213→   213→          garnishId,
   214→   214→          name: typeof c.name === 'string' ? c.name : (menuItem?.name ?? c.menuItemId),
   215→   215→          qty: Math.max(1, Math.min(50, Math.floor(Number(c.qty) || 1))),
   216→   216→          comment: typeof c.comment === 'string' ? c.comment.slice(0, 80) : undefined,
   217→   217→          standalone: c.standalone === true,
   218→   218→        }
   219→   219→      })
   220→   220→    out[tableId] = {
   221→   221→      items,
   222→   222→      vip: value.vip === true,
   223→   223→      tableNote: typeof value.tableNote === 'string' ? value.tableNote.slice(0, 140) : '',
   224→   224→    }
   225→   225→  }
   226→   226→  return out
   227→   227→}

   229→   229→function sanitizeOutbox(raw: unknown): OutboxEntry[] {
   230→   230→  if (!Array.isArray(raw)) return []
   231→   231→  return raw.filter(
   232→   232→    (e): e is OutboxEntry =>
   233→   233→      e &&
   234→   234→      typeof e === 'object' &&
   235→   235→      typeof e.clientOrderId === 'string' &&
   236→   236→      VALID_TABLE_IDS.has(String(e.tableId)) &&
   237→   237→      Array.isArray(e.items) &&
   238→   238→      e.items.length > 0,
   239→   239→  )
   240→   240→}
